    // === 動態生成雜訊紋理 ===
    function generateNoiseTexture() {
        const canvas = document.createElement('canvas');
        canvas.width = 256;
        canvas.height = 256;
        const ctx = canvas.getContext('2d');
        const imgData = ctx.createImageData(256, 256);
        for (let i = 0; i < imgData.data.length; i += 4) {
            // 黑白相間或雜訊感
            const val = Math.random() > 0.5 ? 20 : 60;
            imgData.data[i] = val;     // R
            imgData.data[i+1] = val;   // G
            imgData.data[i+2] = val;   // B
            imgData.data[i+3] = 255;   // A
        }
        ctx.putImageData(imgData, 0, 0);
        const texture = new THREE.CanvasTexture(canvas);
        texture.wrapS = THREE.RepeatWrapping;
        texture.wrapT = THREE.RepeatWrapping;
        texture.repeat.set(2, 2);
        return texture;
    }

    // === DFS 迷宮生成演算法 (產生 15x15 的封閉迷宮) ===
    function generateMazeData() {
        // 初始化全為牆 (1)
        for (let i = 0; i < mapSize; i++) {
            mazeGrid[i] = [];
            for (let j = 0; j < mapSize; j++) {
                mazeGrid[i][j] = 1;
            }
        }

        // 使用 Recursive Backtracker 演算法
        function carve(r, c) {
            mazeGrid[r][c] = 0;
            const dirs = [ [1,0], [-1,0], [0,1], [0,-1] ];
            dirs.sort(() => Math.random() - 0.5); // 打亂方向

            for (let i = 0; i < dirs.length; i++) {
                const nr = r + dirs[i][0] * 2;
                const nc = c + dirs[i][1] * 2;
                if (nr > 0 && nr < mapSize - 1 && nc > 0 && nc < mapSize - 1 && mazeGrid[nr][nc] === 1) {
                    mazeGrid[r + dirs[i][0]][c + dirs[i][1]] = 0; // 打通中間的牆
                    carve(nr, nc);
                }
            }
        }

        // 從 (1,1) 開始生成
        carve(1, 1);
    }
